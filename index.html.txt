    <script>
        // Función sencilla para actualizar el texto del botón al seleccionar un archivo
        function mostrarNombre(input) {
            if (input.files && input.files[0]) {
                const nombreArchivo = input.files[0].name;
                document.querySelector('.file-custom-btn').innerHTML = `✅ Seleccionado: <strong>${nombreArchivo}</strong>`;
                document.querySelector('.file-custom-btn').style.borderStyle = 'solid';
                document.querySelector('.file-custom-btn').style.borderColor = '#10b981';
                document.querySelector('.file-custom-btn').style.color = '#10b981';
            }
        }
    </script>
